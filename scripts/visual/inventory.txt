# Inventory HUD: take the ID out of its slot into the left hand, step west and drop it (Q), step
# back, pick it up again by clicking it, swap hands (X); screenshots of each state.
wait_connected timeout=60
wait_until controlled timeout=30
wait_secs 1                 # sprites, atlases and the HUD settle
screenshot "spawn.png"
click_ui "hud_id"           # empty active hand + ID slot: the ID goes to the hand
wait_secs 0.5
screenshot "id_in_hand.png"
tap A                       # one tile west
wait_secs 0.5
tap Q
wait_secs 0.3
tap D                       # and back
wait_secs 0.5
screenshot "dropped.png"
click_tile -1 0             # the card on the tile just left
wait_secs 0.5
screenshot "picked_up.png"
tap X
wait_secs 0.5
screenshot "swapped.png"
log_state
quit
